// Tab Switching Logic function switchTab(tabId) { document.querySelectorAll('.tab-page').forEach(page => page.classList.add('hidden')); document.getElementById(`tab-content-${tabId}`)?.classList.remove('hidden'); // Update active tab button styles document.querySelectorAll('.nav-tab').forEach(btn => { btn.classList.remove('text-emerald-400', 'bg-emerald-950/80', 'border', 'border-emerald-500/30'); btn.classList.add('text-slate-300'); }); const activeBtn = document.getElementById(`tab-btn-${tabId}`); if (activeBtn) { activeBtn.classList.add('text-emerald-400', 'bg-emerald-950/80', 'border', 'border-emerald-500/30'); activeBtn.classList.remove('text-slate-300'); } } // Mobile Navigation Toggle function toggleMobileMenu() { const menu = document.getElementById('mobile-menu'); menu.classList.toggle('hidden'); } // Modal View Handler function openGardenModal(zoneKey) { // Logic to show modal overlay with specific zone details console.log("Opening garden details for:", zoneKey); } // Filter Cards Handler function filterGardens(category) { const cards = document.querySelectorAll('.garden-card'); cards.forEach(card => { if (category === 'all' || card.getAttribute('data-sun') === category) { card.style.display = 'flex'; } else { card.style.display = 'none'; } }); }
